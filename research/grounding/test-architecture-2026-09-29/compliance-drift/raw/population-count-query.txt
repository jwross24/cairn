from datetime import datetime, timezone
from pathlib import Path
root = Path('/Users/jwross/Documents/cairn/beads_compliance_audit/passes')

if not files:
    raise SystemExit('no spec.json files matched */beads/*/spec.json')

def primary_test_items(paths):
    found = []
    for path in paths:
        spec = json.loads(path.read_text())
        tests = spec.get('checklist', {}).get('tests', {})
        if not isinstance(tests, dict):
            continue
        for items in tests.values():
            if not isinstance(items, list):
                continue
            for item in items:
                if isinstance(item, dict) and str(item.get('id', '')).rsplit('.', 1)[-1] == 'primary':
                    found.append(bool(item.get('expected_path_hints')))
    return found
all_primary = primary_test_items(files)
